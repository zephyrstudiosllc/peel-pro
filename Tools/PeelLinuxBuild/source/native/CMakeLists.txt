cmake_minimum_required(VERSION 3.20)
project(PeelProLinux VERSION 2.0.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This target requires a real Linux sysroot and GTK/PipeWire runtime")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET gtk4>=4.8 gio-unix-2.0 libportal>=0.8 libportal-gtk4>=0.8 gstreamer-1.0>=1.22 gstreamer-app-1.0>=1.22 gstreamer-video-1.0>=1.22 json-glib-1.0 libsecret-1)
add_library(peel_linux SHARED src/app.c src/media.c src/secrets.c src/encoder.c src/h264_annexb.c src/host_ui.c src/discovery.c src/audio.c src/viewer_controls.c)
target_include_directories(peel_linux PUBLIC include)
target_link_libraries(peel_linux PRIVATE PkgConfig::DEPS)
target_compile_options(peel_linux PRIVATE -Wall -Wextra -Wformat=2 -Werror=implicit-function-declaration)
add_executable(peel-linux-native-preview src/main.c)
target_link_libraries(peel-linux-native-preview PRIVATE peel_linux)
install(TARGETS peel_linux LIBRARY DESTINATION lib/peel-pro)
install(TARGETS peel-linux-native-preview RUNTIME DESTINATION lib/peel-pro)

include(CTest)
if(BUILD_TESTING)
  add_executable(peel-linux-encoder-tests tests/encoder_tests.c)
  target_include_directories(peel-linux-encoder-tests PRIVATE src include)
  target_link_libraries(peel-linux-encoder-tests PRIVATE peel_linux PkgConfig::DEPS)
  target_compile_options(peel-linux-encoder-tests PRIVATE -Wall -Wextra -Wformat=2 -Werror=implicit-function-declaration)
  add_test(NAME peel-linux-encoder-seam COMMAND peel-linux-encoder-tests)
  set_tests_properties(peel-linux-encoder-seam PROPERTIES TIMEOUT 20
    ENVIRONMENT "G_MESSAGES_DEBUG=PeelLinuxEncoder;GST_DEBUG_NO_COLOR=1;GST_DEBUG=2,openh264enc:4,h264parse:3,videoencoder:3,appsrc:3")
  # Receiver PCM sink bounds/fences with a substituted non-device sink; no GTK/display/audio server.
  add_executable(peel-linux-audio-tests tests/audio_tests.c)
  target_include_directories(peel-linux-audio-tests PRIVATE src include)
  target_link_libraries(peel-linux-audio-tests PRIVATE peel_linux PkgConfig::DEPS)
  target_compile_options(peel-linux-audio-tests PRIVATE -Wall -Wextra -Wformat=2 -Werror=implicit-function-declaration)
  add_test(NAME peel-linux-audio-sink COMMAND peel-linux-audio-tests)
  set_tests_properties(peel-linux-audio-sink PROPERTIES TIMEOUT 20 ENVIRONMENT "GST_DEBUG_NO_COLOR=1;GST_DEBUG=2,appsrc:3")
endif()
